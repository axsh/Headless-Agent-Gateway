```go
// Package doc comment (標準 examples スタイル)
// artifact-pipeline demonstrates the full artifact lifecycle:
//   1. Upload a local file as a user artifact.
//   2. Retrieve its content and send it to a coding agent with a generation prompt.
//   3. List the files generated by the agent in the session and download one.
//
// Prerequisites:
//   - A running tern server (e.g., via examples/minimal-server)
//   - A configured coding agent (default: wayfinder)
//
// Usage:
//
//	go run . [options]
//
// Examples:
//
//	# Upload input.txt, ask agent to generate output.txt, then download it
//	go run . --input input.txt --key inputs/template.txt --output-key output.txt
//
//	# Use claudecode agent on a remote server
//	go run . --server http://192.168.1.100:3100 --agent claudecode --input data.csv
```

**runUpload()**:
```go
func runUpload(ctx context.Context, c *client.Client, localPath, key string) error {
    resp, err := c.UserArtifacts().PutFile(ctx, key, localPath)
    if err != nil {
        return err
    }
    fmt.Printf("[Step 1] Uploaded: %s (%d bytes, sha256: %s, status: %s)\n",
        resp.Key, resp.Size, resp.SHA, resp.Status)
    return nil
}
```

**runGenerate()**:
```go
func runGenerate(ctx context.Context, c *client.Client, key, agent, model, workDir string) (string, error) {
    // 2a: アーティファクト内容を取得
    rc, err := c.UserArtifacts().Download(ctx, key)
    if err != nil {
        return "", fmt.Errorf("download artifact %q: %w", key, err)
    }
    defer rc.Close()
    content, err := io.ReadAll(rc)
    if err != nil {
        return "", fmt.Errorf("read artifact content: %w", err)
    }
    fmt.Printf("[Step 2] Artifact content fetched: %d bytes\n", len(content))

    // 2b: セッション作成
    session, err := c.CreateSession(ctx, client.SessionRequest{
        Agent:   agent,
        Model:   model,
        WorkDir: workDir,
    })
    if err != nil {
        return "", fmt.Errorf("create session: %w", err)
    }
    defer session.Terminate(ctx)
    fmt.Printf("[Step 2] Session created: %s\n", session.ID)

    // 2c: プロンプト送信
    prompt := fmt.Sprintf(
        "The following is the content of the uploaded artifact %q.\n\n"+
            "---\n%s\n---\n\n"+
            "Based on this content, create a new file called output.txt that summarizes the key points.",
        key, string(content))
    stream, err := session.SendText(ctx, prompt)
    if err != nil {
        return "", fmt.Errorf("send message: %w", err)
    }
    fmt.Println("[Step 2] Agent response:")
    if err := stream.Output(os.Stdout); err != nil {
        return "", fmt.Errorf("stream output: %w", err)
    }

    return session.ID, nil
}
```

**runDownload()**:
```go
func runDownload(ctx context.Context, c *client.Client, sessionID, outputKey, savePath string) error {
    // 3a: セッション内の生成ファイル一覧を取得
    page, err := c.SystemArtifacts().List(ctx, client.SystemArtifactFilter{
        SessionIDs: []string{sessionID},
        Operation:  "create",
    })
    if err != nil {
        return fmt.Errorf("list system artifacts: %w", err)
    }
    fmt.Printf("[Step 3] Generated files in session %s (%d total):\n", sessionID, page.TotalCount)
    for _, item := range page.Items {
        fmt.Printf("  %s  (%s)\n", item.Key, item.Operation)
    }

    // 3b: ダウンロード対象キーを決定
    target := outputKey
    if target == "" {
        if len(page.Items) == 0 {
            return fmt.Errorf("no generated files found in session %s", sessionID)
        }
        target = page.Items[0].Key
    }

    // 3c: ファイルダウンロード
    rc, err := c.SystemArtifacts().Download(ctx, target)
    if err != nil {
        return fmt.Errorf("download %q: %w", target, err)
    }
    defer rc.Close()

    data, err := io.ReadAll(rc)
    if err != nil {
        return fmt.Errorf("read downloaded content: %w", err)
    }

    // キーからファイル名を導出（savePath が空の場合）
    if savePath == "" {
        parts := strings.Split(target, "/")
        savePath = "./" + parts[len(parts)-1]
    }

    if err := os.WriteFile(savePath, data, 0o644); err != nil {
        return fmt.Errorf("write file %q: %w", savePath, err)
    }
    fmt.Printf("[Step 3] Downloaded: %s -> %s (%d bytes)\n", target, savePath, len(data))
    return nil
}
```
